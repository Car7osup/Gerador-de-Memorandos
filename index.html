// CÓDIGO DO SEU SERVIDOR INVISÍVEL const GEMINI_API_KEY = "COLE_SUA_CHAVE_AQUI"; function doPost(e) { try { const dadosRecebidos = JSON.parse(e.postData.contents); const textoOriginal = dadosRecebidos.texto; const url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=" + GEMINI_API_KEY; const prompt = `Aja como um assistente administrativo da Secretaria de Educação. O usuário enviou uma frase coloquial informando um problema ou serviço necessário. Transforme a frase em um pedido formal e direto para constar em um memorando de ofício. Retorne APENAS um JSON válido contendo exatamente duas chaves: "nome": Um título curto do serviço em MAIÚSCULAS. "descricao": A descrição formal e detalhada do pedido. Sem crases, sem introdução. Texto: "${textoOriginal}"`; const options = { method: "post", contentType: "application/json", payload: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }; const respostaGemini = UrlFetchApp.fetch(url, options); const jsonResult = JSON.parse(respostaGemini.getContentText()).candidates[0].content.parts[0].text.trim().replace(/```json/gi, '').replace(/```/g, '').trim(); return ContentService.createTextOutput(jsonResult).setMimeType(ContentService.MimeType.JSON); } catch (erro) { return ContentService.createTextOutput(JSON.stringify({ erro: erro.message })).setMimeType(ContentService.MimeType.JSON); } } // Isso é obrigatório para não dar erro de bloqueio (CORS) function doOptions(e) { return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); }
